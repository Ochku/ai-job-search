%-----------------%
% Custom Commands %
%-----------------%

% Header: name, plain-text contact line (no icon fonts: ATS parsers read icon names as words), location line
\newcommand{\documentHeader}[3]{
  \begin{center}
    {\Huge\scshape\color{accentTitle} #1\par}
    \vspace{4pt}
    {\color{accentLine} \hrule}
    \vspace{3pt}
    {\small #2\par}
    \vspace{1pt}
    {\small #3\par}
    \vspace{3pt}
    {\color{accentLine} \hrule}
  \end{center}
  \vspace{-4pt}
}

\newcommand{\sep}{\unskip\enspace\textperiodcentered\enspace\ignorespaces} % \unskip/\ignorespaces stop source line breaks adding extra space

\titleformat{\section}{
  \vspace{2pt}
  \color{accentText}
  \scshape\raggedright\large\bfseries
}{}{0em}{}[\color{accentLine}\titlerule]
\titlespacing*{\section}{0pt}{4pt}{3pt}

% Entry: company/institution (bold) with dates right-aligned, then title(s) in italics
\newcommand{\entry}[3]{
  \textbf{#1}\hfill\textbf{#2}\\
  \textit{#3}\par
}

% Sub-group label inside one employer's bullets
\newcommand{\subgroup}[1]{
  \vspace{2pt}
  {\small\bfseries\color{accentText}#1}\par
}

\newenvironment{resume_list}{
  \vspace{-1pt}
  \begin{itemize}[itemsep=2pt, parsep=1pt, topsep=1pt, leftmargin=14pt, labelsep=6pt]
}{
  \end{itemize}
  \vspace{2pt}
}

\renewcommand\labelitemi{\textbullet}
