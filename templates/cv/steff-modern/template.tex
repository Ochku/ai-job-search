% \cvheader{name}{headline}{contact line}
\newcommand{\cvheader}[3]{%
  {\fontsize{24}{28}\selectfont\bfseries\color{nameColor}#1\par}
  \vspace{2pt}
  {\large\color{accent}#2\par}
  \vspace{3pt}
  {\small\color{muted}#3\par}
  \vspace{2pt}
}

\newcommand{\sep}{\unskip\enspace|\enspace\ignorespaces} % header separator
\newcommand{\dotsep}{\unskip\ \textperiodcentered\ \ignorespaces} % inline separator

% \MakeUppercase sits in the before-code so it only uppercases the title, not the rule's color name
\titleformat{\section}{\color{accent}\bfseries\normalsize}{}{0em}{\MakeUppercase}[\vspace{1pt}{\color{ruleColor}\titlerule[0.6pt]}]
\titlespacing*{\section}{0pt}{8pt}{4pt}

% \entry{Company}{Department or title}{dates}
\newcommand{\entry}[3]{%
  \vspace{3pt}
  {\large\textbf{#1}}\ {\color{subAccent}\large\textbf{---\ #2}}\hfill{\color{muted}#3}\par
}
% Title line(s) under an entry
\newcommand{\roles}[1]{{\color{muted}#1\par}}
% Sub-group label inside one employer's bullets
\newcommand{\subgroup}[1]{\vspace{1pt}{\color{muted}\itshape #1\par}}
